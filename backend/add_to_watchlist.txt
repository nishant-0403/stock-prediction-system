@app.post("/watchlist/add")
def add_to_watchlist(req: WatchlistAddRequest, db: Session = Depends(get_db)):

    user = db.execute(text("SELECT user_id FROM User WHERE user_id = :uid"), {"uid": req.user_id}).fetchone()
    if not user:
        raise HTTPException(status_code=404, detail="User not found")

    watchlist = db.execute(
        text("SELECT watchlist_id FROM Watchlist WHERE user_id = :uid"),
        {"uid": req.user_id}
    ).fetchone()

    if not watchlist:
        db.execute(
            text("INSERT INTO Watchlist (user_id) VALUES (:uid)"),
            {"uid": req.user_id}
        )
        db.commit()
        watchlist = db.execute(
            text("SELECT watchlist_id FROM Watchlist WHERE user_id = :uid"),
            {"uid": req.user_id}
        ).fetchone()

    watchlist_id = watchlist[0]

    existing = db.execute(
        text("SELECT * FROM Watchlist_Stock WHERE watchlist_id = :wid AND stock_id = :sid"),
        {"wid": watchlist_id, "sid": req.stock_id}
    ).fetchone()

    if existing:
        raise HTTPException(status_code=409, detail="Stock already in watchlist")

    db.execute(
        text("INSERT INTO Watchlist_Stock (watchlist_id, stock_id, date_added) VALUES (:wid, :sid, NOW())"),
        {"wid": watchlist_id, "sid": req.stock_id}
    )   
    db.commit()

    return {"message": "Stock added successfully", "watchlist_id": watchlist_id}
